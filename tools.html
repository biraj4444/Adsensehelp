5 · Live page inspector

The tools above cannot see your live site, because browsers stop one website reading another. This snippet runs inside your own page instead and reports what AdSense is doing.

  1. Open your live page on a desktop browser.
  2. Open Developer Tools (F12) and choose the Console tab.
  3. Paste the snippet and press Enter.

(function () {
  var s = [].slice.call(document.querySelectorAll('script[src*="adsbygoogle.js"]'));
  var ins = [].slice.call(document.querySelectorAll('ins.adsbygoogle'));
  var meta = document.querySelector('meta[name="google-adsense-account"]');
  var r = ['AdSense script tags: ' + s.length + (s.length > 1 ? '  (more than one: remove duplicates)' : '')];
  s.forEach(function (x) {
    var m = /client=(ca-pub-\d+)/.exec(x.src);
    r.push('  client: ' + (m ? m[1] : 'MISSING') + '  in <' + (x.closest('head') ? 'head' : 'body') + '>');
  });
  r.push('Ad units (ins.adsbygoogle): ' + ins.length);
  ins.forEach(function (i, n) {
    r.push('  #' + (n + 1) + ' processed=' + (i.getAttribute('data-adsbygoogle-status') || 'no') +
      ' ad=' + (i.getAttribute('data-ad-status') || 'n/a') + ' size=' + i.offsetWidth + 'x' + i.offsetHeight);
  });
  if (meta) r.push('Account meta tag: ' + meta.content);
  console.log(r.join('\n'));
})();

Note

It only reads attributes already on your page, and sends nothing anywhere. Android browsers have no Developer Tools console, so use a desktop browser or remote debugging from your phone.

6 · Quick reference

  • Publisher ID: pub- + 16 digits. In HTML: ca-pub- + the same digits.
  • Ad slot ID: a number (usually 10 digits) for one ad unit. Not the same as the publisher ID.
  • ads.txt line: google.com, pub-ID, DIRECT, f08c47fec0942fa0
  • Crawler name: Mediapartners-Google
