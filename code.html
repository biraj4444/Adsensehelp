Version notes

Modern browsers only need the code below. Do not call push more times than there are unfilled ins.adsbygoogle elements.

document.addEventListener("DOMContentLoaded", function () {
  var units = document.querySelectorAll("ins.adsbygoogle:not([data-adsbygoogle-status])");
  units.forEach(function () {
    try {
      (window.adsbygoogle = window.adsbygoogle || []).push({});
    } catch (err) {
      console.error("AdSense push failed:", err);
    }
  });
});

Each push({}) fills one unfilled ins.adsbygoogle element. The selector skips units that already have a status attribute, which prevents the error about elements already having ads. The try/catch stops one failure from breaking your other scripts.

import { useEffect, useRef } from "react";

const CLIENT = import.meta.env.VITE_ADSENSE_CLIENT; // ca-pub-XXXXXXXXXXXXXXXX

export default function AdUnit({ slot, format = "auto" }) {
  const pushed = useRef(false);

  useEffect(() => {
    if (pushed.current) return;
    try {
      (window.adsbygoogle = window.adsbygoogle || []).push({});
      pushed.current = true;
    } catch (err) {
      console.error("AdSense push failed:", err);
    }
  }, []);

  return (
    <ins
      className="adsbygoogle"
      style={{ display: "block" }}
      data-ad-client={CLIENT}
      data-ad-slot={slot}
      data-ad-format={format}
      data-full-width-responsive="true"
    />
  );
}

📍 Where to put this

File
src/components/AdUnit.jsx
Location
New component file; render <AdUnit /> where the ad should appear
Why

Create src/components/AdUnit.jsx, import it in a page component and render <AdUnit slot="1234567890" /> where you want the ad.

your-react-app/├── index.html└── src/    ├── components/    │   └── AdUnit.jsx   ← ADD CODE HERE    └── App.jsx   ← render <AdUnit /> here

Understand it

useEffect runs after the <ins> element is in the DOM, then asks AdSense to fill it. The useRef guard stops React 18+ StrictMode (development only) from calling push twice, which would log: "All ins elements ... already have ads in them".

Version difference

const http = require("http");
const fs = require("fs");
const path = require("path");

const CLIENT = process.env.ADSENSE_CLIENT || "";
const VALID = /^ca-pub-\d{16}$/.test(CLIENT);

const adsenseTag = VALID
  ? `<script async src="https://pagead2.googlesyndication.com/pagead/js/adsbygoogle.js?client=${CLIENT}" crossorigin="anonymous"></script>`
  : "";

http.createServer((req, res) => {
  if (req.url === "/ads.txt") {
    res.writeHead(200, { "Content-Type": "text/plain; charset=utf-8" });
    return res.end(fs.readFileSync(path.join(__dirname, "ads.txt")));
  }
  res.writeHead(200, { "Content-Type": "text/html; charset=utf-8" });
  res.end(`<!doctype html>
<html lang="en">
<head>
  <meta charset="utf-8">
  <title>My Node site</title>
  ${adsenseTag}
</head>
<body><h1>Hello</h1></body>
</html>`);
}).listen(process.env.PORT || 3000);
